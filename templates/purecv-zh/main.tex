\newcommand{\resumeZhRow}[2]{%
  \par\noindent
  \if\relax\detokenize{#2}\relax
    #1\par
  \else
    \sbox{\resumeZhRowBox}{%
      \begin{tabularx}{\linewidth}{@{}>{\raggedright\arraybackslash}X@{\hspace{1em}}>{\raggedleft\arraybackslash}p{0.28\linewidth}@{}}
        #1 & #2 \\
      \end{tabularx}%
    }%
    % Tall tabular rows cannot split across pages. Measure the real fonts/width,
    % then use normal breakable paragraphs rather than shipping an oversized box.
    \ifdim\dimexpr\ht\resumeZhRowBox+\dp\resumeZhRowBox\relax>0.5\textheight
      #1\par
      {\raggedleft #2\par}%
    \else
      \usebox{\resumeZhRowBox}\par
    \fi
  \fi
}
